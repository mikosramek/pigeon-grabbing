void change<LEVEL_NAME>Scene(uint32_t sceneID, uint32_t spawnPointID) {
  // unload active scene
  if (<LEVEL_NAME>Level.activeScene != NULL) {
    switch (<LEVEL_NAME>Level.activeScene->id) {
      <UNLOAD>
      default: assertf(false, "Current scene doesn't have an unload: %d", (int)<LEVEL_NAME>Level.activeScene->id);
    }
  }

  // load target scene
  switch (sceneID) {
    <LOAD>
    default: assertf(false, "Trying to load a scene that doesn't exist: %d", (int)sceneID);
  }

  // find spawn point via ID
  SpawnPoint *spawn = NULL;
  for (int i = 0; i < <LEVEL_NAME>Level.activeScene->spawnCount; i += 1) {
    uint32_t id = <LEVEL_NAME>Level.activeScene->spawns[i].id;
    if (id == spawnPointID) {
      spawn = &<LEVEL_NAME>Level.activeScene->spawns[i];
    }
  }

  assertf(spawn != NULL, "NO SPAWN FOUND!");
  
  // move player to spawnPoint
  Player *p = getPlayer();
  p->position = spawn->spawn;
  p->cameraAngle = spawn->angle;
}
